def audit(event, arguments):
    if event in {'subprocess.Popen', 'os.system', 'os.posix_spawn', 'os.fork', 'os.forkpty', 'os.exec',

def parse(path):
    return ast.parse(path.read_bytes(), filename=str(path))


def assignments(tree, names):
    return [node for node in tree.body if isinstance(node, ast.Assign)
            and any(isinstance(target, ast.Name) and target.id in names for target in node.targets)
            or isinstance(node, ast.AugAssign) and isinstance(node.target, ast.Name) and node.target.id in names]


def execute_fragment(nodes, namespace):
    exec(compile(ast.Module(body=nodes, type_ignores=[]), 'inert-case-inventory-only', 'exec'), namespace)


class Inventory:
    def __init__(self, *_):
        self.names = []
    def case(self, name, callback):
        assert type(name) is str and name not in self.names
        self.names.append(name)
    def track(self):
        return nullcontext()
    def __getattr__(self, name):
        return lambda *_args, **_kwargs: None


def case_inventory(path, mode):
    tree, suite = parse(path), Inventory()
    namespace = {node.name: (lambda *_a, **_k: None) for node in tree.body if isinstance(node, ast.FunctionDef)}
    namespace.update(suite=suite, io=SimpleNamespace(LIMITS=limits), discovery=SimpleNamespace(CLASSIFICATION=classification),
                     case=suite.case, mapping=[])
    if mode == 'original':
        main = next(node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == 'main')
        body = next(node for node in main.body if isinstance(node, ast.Try)).body
    elif mode in {'mapped', 'integration', 'vpg'}:
        if mode == 'vpg':
            names = {target.id for node in tree.body if isinstance(node, ast.Assign) for target in node.targets
                     if isinstance(target, ast.Name) and target.id.endswith('_CASES')}
            execute_fragment(assignments(tree, names), namespace)
        if mode == 'mapped':
            def mapped(old, new, callback, reason=None):
                namespace['mapping'].append({'historicalCase': old, 'newCase': new, 'reason': reason})
                suite.case(new, callback)
            namespace['case'] = mapped
        body = next(node for node in tree.body if isinstance(node, ast.Try)).body
    else:
        main = next(node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == 'main')
        body = []
        for node in main.body:
            if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'report' for t in node.targets):
                break
            body.append(node)
    execute_fragment(body, namespace)
    return suite.names, namespace
