# These proof-fixture guards do not change any action/native/graph limit.
# All cases run sequentially. The manifest reserves this space before each
# fresh fixture; historical failed fixtures are never reclaimed by this driver.
PROOF_FREE_FLOOR = 4 * 1024**3
PROOF_FIXTURE_HEADROOM = 512 * 1024**2


def proof_space(root, *, before):
    usage = shutil.disk_usage(root)
    if type(usage.free) is not int or usage.free < 0 or type(before) is not bool:
        raise RuntimeError('invalid proof disk observation')
    reserve = PROOF_FIXTURE_HEADROOM if before else 0
    return {'freeBytes': usage.free, 'floorBytes': PROOF_FREE_FLOOR,
            'reservedFixtureBytes': reserve,
            'sufficient': usage.free >= PROOF_FREE_FLOOR + reserve,
            'scope': 'cooperative filesystem free-space observation; no cleanup authority'}

def summary_only(ctx):
    # Full actual native records are retained by each case/immutable observation;
    # avoid duplicating their bounded raw pipe contents in aggregate summaries.
    value=ctx.summary()
    value['native']=[{'pid':r['pid'],'ownership':r['ownership'],'status':r['status'],
                      'recordSha256':hashlib.sha256(json.dumps(r,sort_keys=True).encode()).hexdigest()}
                     for r in ctx.native]
    return value

    def case(self, name, callback):
        if self.resource_blocked is not None:
            raise RuntimeError('proof resource stop is terminal; no further case acquisition')
        area = self.root/name
        area.mkdir(mode=0o700)
        before_children, before_signals = len(self.children), len(self.signals)
        started = time.monotonic()
        disk_before, disk_after = None, None
        try:
            disk_before = proof_space(self.root, before=True)
            if not disk_before['sufficient']:
                self.resource_blocked = 'free space below 4GiB plus declared fixture headroom'
                raise RuntimeError(self.resource_blocked)
            result = callback(area)
            row = {'case': name, 'status': 'PASS', 'result': result}
        except BaseException as error:
            if disk_before is None:
                self.resource_blocked = 'pre-fixture disk observation failed: ' + type(error).__name__
            row = {'case': name, 'status': 'FAIL', 'errorType': type(error).__name__,
                    'error': str(error), 'traceback': traceback.format_exc()}
        unresolved = []
        for child in self.children[before_children:]:
            if child.returncode is None:
                try:
                    child.wait(timeout=3)
                except BaseException as error:
                    unresolved.append({'pid': child.pid, 'errorType': type(error).__name__})
        if unresolved:
            row.update(status='FAIL', unresolvedActualChildren=unresolved,
                       nextCases='BLOCKED; no new native acquisition or guessed-PID signaling')
            self.resource_blocked = 'actual child unresolved'
        try:
            disk_after = proof_space(self.root, before=False)
            if not disk_after['sufficient']:
                self.resource_blocked = 'free space below retained 4GiB floor'
        except BaseException as error:
            self.resource_blocked = 'disk observation failed: ' + type(error).__name__
        if self.resource_blocked is not None:
            row.update(status='FAIL', nextCases='BLOCKED; ' + self.resource_blocked)
        row.update(seconds=time.monotonic()-started, directChildren=len(self.children)-before_children,
                   signalRequests=self.signals[before_signals:],
                   diskBefore=disk_before, diskAfter=disk_after)
        self.results.append(row)
        with io.action('record-proof-'+name) as ctx:
            io.write_new(area/'RESULT.json', row, ctx=ctx)
        print(row['status'], name, flush=True)
        if self.resource_blocked is not None:
            raise RuntimeError(self.resource_blocked + '; stop further cases')
        return row

    @contextmanager
    def track(self):
        suite = self
        class Child(self.real_popen):
            def __init__(self, *args, **kwargs):
                super().__init__(*args, **kwargs)
                suite.children.append(self)
        def stop(pid, number):
            matches = [child for child in self.children if child.pid == pid and child.returncode is None]
            if pid == os.getpid():
                return self.real_kill(pid, number)  # explicit fixture cancellation, never foreign process
            assert len(matches) == 1, 'VETO unowned/reaped direct signal'
            self.signals.append({'pid': pid, 'signal': number, 'kind': 'direct', 'reserved': True})
            return self.real_kill(pid, number)
        def group(pid, number):
            matches = [child for child in self.children if child.pid == pid and child.returncode is None]
            assert len(matches) == 1, 'VETO historical/unowned group signal'
            self.signals.append({'pid': pid, 'signal': number, 'kind': 'reserved-outer-query', 'reserved': True})
            return self.real_killpg(pid, number)
        with patch.object(subprocess, 'Popen', Child), patch.object(os, 'kill', stop), patch.object(os, 'killpg', group):
            yield

    def query(self, code, *, seconds=2, cap=pg.PS_BYTES):
        with patch.object(pg, 'PS_COMMAND', [self.python, '-I', '-B', '-u', '-c', code]):
            record, rows = pg.numeric_snapshot(seconds=seconds, byte_limit=cap)
        assert record['ownership'] == 'reaped', record
        return record, rows

    def state(self, area, state, expected):
        query, rows = self.query(f"print('12341 1 12340 {os.getuid()} {state}')")
        if expected == 'MALFORMED':
            assert rows is None and query['status'] == 'UNKNOWN'
            return {'query': query, 'rows': rows}
        assert rows is not None and query['status'] == 'PARSED'
        value = pg.classify_group(rows, group=12340, uid=os.getuid())
        assert value['status'] == expected, value
        return {'query': query, 'classification': value}

    def query_bad(self, area, code, *, seconds=2, cap=pg.PS_BYTES):

        assert rows is None and query['status'] == 'UNKNOWN'
        return query

    def native(self, area, code, *, expected=True, stdout=nr.LIMITS['stdout'], stderr=nr.LIMITS['stderr'], seconds=2, input_data=b'', frames=None):
        ctx = None
        try:
            with io.action('native-proof', seconds=min(5, max(seconds+0.5, 1))) as ctx:
                if frames is not None:
                    frames = nr.BatchFrames(frames, ctx=ctx)
                out, err, record = nr.run([self.python, '-I', '-B', '-u', '-c', code], cwd=area,
                                           environment={'PATH': '/usr/bin:/bin', 'LC_ALL': 'C'}, ctx=ctx,
                                           stdout_limit=stdout, stderr_limit=stderr, seconds=seconds,
                                           input_data=input_data, frames=frames, allow_stderr=expected)
            assert expected, 'dangerous native case accepted'
            return {'record': record, 'outBytes': len(out), 'errBytes': len(err), 'action': summary_only(ctx)}
        except nr.NativeFailure as error:
            assert not expected, str(error)
            assert error.record['ownership'] == 'reaped', error.record
            assert not error.record['signalAuthority']
            return {'rejected': str(error), 'record': error.record, 'action': summary_only(ctx)}

    def frame(self, area, variant):
        body = b'synthetic body\n'
        oid = hashlib.sha1(b'blob '+str(len(body)).encode()+b'\0'+body).hexdigest()
        record = {'oid': oid, 'type': 'blob', 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()}
        header = f'{oid} blob {len(body)}\n'.encode()
        output = header+body+b'\n'
        code_tail = ''
        if variant == 'header-long': output = b'x'*97
        if variant == 'wrong-oid': output = b'0'*40+output[40:]
        if variant == 'wrong-type': output = output.replace(b' blob ', b' tree ', 1)
        if variant == 'large-size': output = header.replace(str(len(body)).encode(), b'999999999')
        if variant == 'noncanonical-size': output = header.replace(str(len(body)).encode(), b'0'+str(len(body)).encode())
        if variant == 'short': output = output[:-4]
        if variant == 'delimiter': output = output[:-1]+b'x'
        if variant == 'extra': output += b'x'
        if variant == 'nonzero': code_tail = ';raise SystemExit(7)'
        if variant == 'stderr': code_tail = ";os.write(2,b'diagnostic')"
        code = f"import os;os.read(0,41);os.write(1,{output!r})"+code_tail
        return self.native(area, code, expected=variant == 'valid', frames=[record])

    def fs(self, area, mode):
        path = area/'value'
        if mode.startswith('json-'):
            data = {'duplicate': b'{"x":1,"x":2}', 'constant': b'{"x":NaN}',
                    'depth': b'['*65+b'0'+b']'*65, 'string': b'"'+b'x'*(io.LIMITS['jsonString']+1)+b'"',
                    'bytes': b' '*(io.LIMITS['record']+1), 'tokens': b'['+b'0,'*270000+b'0]',
                    'valid': b'{"valid":[1,true,null,"ok"]}'}[mode[5:]]
            path.write_bytes(data)
            ctx = None
            try:
                with io.action('json-proof') as ctx:
                    value = io.read_json(path, ctx=ctx)
                assert mode == 'json-valid', 'bad JSON accepted'
                return {'parsed': value, 'counts': ctx.counts}
            except (RuntimeError, ValueError):
                assert mode != 'json-valid'
                return {'rejected': True, 'counts': ctx.counts}
        if mode in ('at-limit','over-limit','tool-large'):
            size = io.LIMITS['blob']+(1 if mode == 'over-limit' else 1024 if mode == 'tool-large' else 0)
            with path.open('xb') as stream:
                for _ in range(size//io.CHUNK): stream.write(b'x'*io.CHUNK)
                stream.write(b'x'*(size%io.CHUNK))
            ctx = None
            try:
                with io.action('digest-proof') as ctx:
                    value=io.digest(path, ctx=ctx, role='tool' if mode == 'tool-large' else 'source')
                assert mode!='over-limit'; return {'sha256':value,'counts':ctx.counts}
            except RuntimeError:
                assert mode=='over-limit'; return {'rejected':True,'counts':ctx.counts}
        if mode in ('expired','pending-cancel','cumulative'):
            path.write_bytes(b'real-data')
            try:
                with io.action('budget-proof',seconds=0.025 if mode=='expired' else 5) as ctx:
                    if mode=='expired':time.sleep(0.035)
                    if mode=='pending-cancel':ctx.pending.signal=signal.SIGTERM
                    if mode=='cumulative':ctx.counts['allBytes']=io.LIMITS['allBytes']-12
                    io.digest(path,ctx=ctx,role='source')
            except RuntimeError:return {'rejected':True,'counts':ctx.counts,'failure':ctx.failure}
            raise AssertionError('exhausted/cancelled action performed ordinary work')
        raise AssertionError('unknown fs scenario')

    def deletion(self, area, variant):
        with io.action('deletion-setup') as ctx:
            owner=io.create_owned(area/'owned',area/'OWNER.json','0'*40,'synthetic deletion proof',ctx=ctx)
            for name in ('a','b','c'):(area/'owned'/name).write_bytes(b'SYNTHETIC '+name.encode())
            before=io.inventory(owner,ctx=ctx)
        flag=pg.Interrupted();ctx=None
        real_unlink=os.unlink
        calls=[]
        def remove(name,*args,**kwargs):
            result=real_unlink(name,*args,**kwargs);calls.append(name)
            if len(calls)==1 and variant=='cancel-partial':flag.signal=signal.SIGTERM
            if len(calls)==1 and variant=='replace-partial':
                (area/'owned'/'a').rename(area/'preserved-a')
                (area/'owned'/'a').write_bytes(b'FOREIGN SUBSTITUTION')
            return result
        if variant=='cancel-before':flag.signal=signal.SIGINT
        if variant=='extra-before':(area/'owned'/'unknown').write_bytes(b'FOREIGN')
        try:
            with io.action('deletion-proof',pending=flag,seconds=10) as ctx:
                with patch.object(os,'unlink',remove):io.delete_inventory(owner,before,ctx=ctx)
            assert variant=='positive'
            return {'removed':len(ctx.removals),'rootGone':not (area/'owned').exists()}
        except RuntimeError:
            assert variant!='positive'
            if variant in ('cancel-before','extra-before'):assert not calls
            if variant=='replace-partial':assert (area/'owned'/'a').read_bytes()==b'FOREIGN SUBSTITUTION'
            assert (area/'owned').exists()
            return {'failure':ctx.failure,'removals':ctx.removals,'calls':calls,'retained':sorted(p.name for p in (area/'owned').iterdir())}

    def graph(self, area, variant):
        root,head,base,files=self.repo(area)
        if variant=='modified':(root/'alpha.txt').write_text('modified clean\n')
        elif variant=='deleted':(root/'alpha.txt').unlink()
        elif variant=='untracked':(root/'added.txt').write_text('new clean\n')
        elif variant=='executable':(root/'alpha.txt').chmod(0o755)
        elif variant=='empty-file':(root/'empty.txt').write_bytes(b'')
        elif variant=='nested':(root/'nested/deeper').mkdir();(root/'nested/deeper/new.txt').write_text('new\n')
        elif variant.startswith('whitespace'):
            target=root/('new.txt' if 'new' in variant else 'alpha.txt')
            target.write_text('bad trailing \n' if 'blank-eof' not in variant else 'value\n\n')
        elif variant=='hostile-attributes':
            (root/'.gitattributes').write_text('* -whitespace diff=missing textconv=missing\n')
            (root/'alpha.txt').write_text('bad trailing \n')
            (root/'.git/config').write_text('[include]\n path = /nonexistent/mrk-config\n[core]\n fsmonitor = /nonexistent/mrk-helper\n whitespace = -blank-at-eol\n[diff "missing"]\n command = /nonexistent/mrk-diff\n textconv = /nonexistent/mrk-textconv\n')
        before={str(p.relative_to(root/'.git')):hashlib.sha256(p.read_bytes()).hexdigest() for p in (root/'.git').rglob('*') if p.is_file()}
        ctx=None
        try:
            with io.action('real-git-proof') as ctx:
                raw=git.raw_source(root,branch='fixture',expected_gitdir=root/'.git',expected_common=root/'.git',ctx=ctx)
                graph=raw['graph']
                approved=dict(files)
                for name in (set(files) | {'added.txt','empty.txt','nested/deeper/new.txt','new.txt','.gitattributes'}):
                    if (root/name).exists():
                        body=io.read_file(root/name,ctx=ctx,role='source')
                        approved[name]={'sha256':body['sha256'],'executable':bool(body['identity']['mode']&0o100)}
                    elif name in files:approved[name]={'deleted':True}
                accepted={'role':'execution','structuralDirectories':sorted({str(p) for n in approved for p in Path(n).parents if str(p)!='.'}),
                          'additionalControls':{},'agents':None}
                universe,bodies=discovery.discover(root,raw['baseFiles'],accepted,approved,ctx=ctx)
                owner,view=git.create_view(area/'view',area/'OWNER.json',graph,tree=base,ctx=ctx,index=raw['indexBytes'],anchor=raw['anchorBytes'],
                                           universe=universe,control_bodies=bodies)
                logical,physical=git.source_files(root,raw,owner,self.leaf,ctx=ctx,accepted=approved,universe=universe)
                entries=git.validate_view(owner,graph,ctx=ctx,with_index=True,expected_inventory=view['initialInventory'],universe=universe,mirror=view['mirror'])
                git.remove_view(owner,view,entries,ctx=ctx)
                # Fresh final forest contains no commit anchor or copied index.
                forest=git.Forest(ctx)
                for oid,(record,body) in graph.objects.items():
                    if record['type']!='commit':forest.add(record['type'],body,expected=oid)
                proposed,forest=git.proposed_tree(root,logical,ctx=ctx,graph=forest)
                owner,view=git.create_view(area/'diff',area/'DIFF-OWNER.json',forest,tree=proposed,ctx=ctx)
                native=git.consume_forest(owner,forest,base,proposed,self.leaf,ctx=ctx)
                git.remove_view(owner,view,native['inventory'],ctx=ctx)
                git.recheck_raw(raw,ctx=ctx)
                assert not variant.startswith('whitespace') and variant!='hostile-attributes','whitespace suppression accepted'
                result={'base':base,'proposed':proposed,'files':logical,'native':native,'action':summary_only(ctx)}
        except nr.NativeFailure as error:
            assert variant.startswith('whitespace') or variant=='hostile-attributes',str(error)
            result={'expectedWhitespaceFailure':error.record,'action':summary_only(ctx)}
        after={str(p.relative_to(root/'.git')):hashlib.sha256(p.read_bytes()).hexdigest() for p in (root/'.git').rglob('*') if p.is_file()}
        assert before==after,'real fixture Git store/index/ref changed'
        return result

    def clone_actual_repo(self, area, original):
        """Fixture copies of actual tracked/included bytes, never real Git writes."""
        with io.action('actual-source-applicability-copy') as ctx:
            config=io.read_json(self.evidence/'CONFIG.json',ctx=ctx)
            ctx.check()
            temporary=Path(tempfile.mkdtemp(prefix='.mrk-qa006-verifier-r3-',dir=str(Path(config['owner']).parent)))
            ctx.check()
            assert not any(temporary.is_relative_to(Path(root)) for root in config['gitLayouts'])
            temporary_owner={'path':str(temporary),'identity':io.identity(temporary,ctx=ctx),
                             'parentAncestry':io.ancestry(temporary.parent,ctx=ctx),'nonce':os.urandom(24).hex(),
                             'tree':config['expectedSourceHead'],'purpose':'actual-source preparation tmp outside both repositories; actual mkdtemp reservation'}
            io.write_new(area/'PREPARATION-TMP-OWNER.json',temporary_owner,ctx=ctx)
            layout=config['gitLayouts'][original]
            raw=git.raw_source(original,branch=layout['branch'],expected_gitdir=layout['gitdir'],expected_common=layout['common'],ctx=ctx)
            acceptance,original_logical=discovery.load_acceptance(self.evidence,config,ctx=ctx)
            # This input is never executed. Excluded live-source caches remain
            # untouched; the fresh execution clone contains accepted files only.
            accepted={**acceptance['record']['roots'][original],'role':'preservation'}
            universe,bodies=discovery.discover(original,raw['baseFiles'],accepted,original_logical[original]['files'],ctx=ctx)
            owner,view=git.create_view(temporary/'clone-view',area/'CLONE-VIEW-OWNER.json',raw['graph'],tree=raw['stable']['baseRoot'],
                                     index=raw['indexBytes'],anchor=raw['anchorBytes'],ctx=ctx,universe=universe,control_bodies=bodies)
            files,physical=git.source_files(original,raw,owner,self.leaf,ctx=ctx,accepted=original_logical[original]['files'],universe=universe)
            entries=git.validate_view(owner,raw['graph'],with_index=raw['indexBytes'] is not None,ctx=ctx,expected_inventory=view['initialInventory'],
                                      universe=universe,mirror=view['mirror'])

    def native_fault(self, area, variant):
