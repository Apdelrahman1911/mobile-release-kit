class NativeBoundaryViolation(BaseException):
    pass


def denied(label):
    def veto(*_args, **_kwargs):
        NATIVE_ATTEMPTS.append(label)
        raise NativeBoundaryViolation(label)
    return veto


def preparation_bound():
    if REAL_MONOTONIC() >= PURE_DEADLINE:
        raise NativeBoundaryViolation("original-pure-60s-cutoff")
    if NATIVE_ATTEMPTS:

def native_audit(event, _args):

                 "os.system", "ctypes.dlopen", "socket.__new__", "socket.connect", "socket.bind"}:
        denied("audit:" + event)()

class BoundedOutput:
    retained = 0

    def __init__(self, stream):
        self.stream = stream

    def write(self, text):
        preparation_bound()
        amount = len(text.encode("utf-8"))
        if type(self).retained + amount > 2 * 1024 * 1024:
            raise NativeBoundaryViolation("combined-pure-output-cap")
        type(self).retained += amount
        result = self.stream.write(text)
        preparation_bound()
        return result

    def flush(self):
        return self.stream.flush()


sys.stdout, sys.stderr = BoundedOutput(sys.stdout), BoundedOutput(sys.stderr)


def definitions(name):
    path = BASE / (name + ".py")
    parsed = ast.parse(path.read_bytes(), str(path))
    allowed = (ast.Import, ast.ImportFrom, ast.Assign, ast.AnnAssign, ast.FunctionDef, ast.ClassDef)
    parsed.body = [node for node in parsed.body if isinstance(node, allowed)]
    module = types.ModuleType(name)
    module.__file__ = str(path)
    sys.modules[name] = module
    exec(compile(parsed, str(path), "exec"), module.__dict__)
    return module

def admission(c, actor, ident):
    # Fixture is an explicit synthetic retained record, not a call to its issuer.
    return dict(version=1, actor=actor, identity=copy.deepcopy(ident), contextSha256=hashlib.sha256(C.dumps(c)).hexdigest(),
                admittedAt=101.0, work=c["work"], disposition=c["disposition"], outer=c["outer"],
                controlRole="C" if actor == "sender" and C.ROUTES[c["row"]].loss else "stdin",
                controlEof=False, controlFailed=None, cancelled=False)


def capture(data):
    return dict(eof=True, observedBytes=len(data), retainedBytes=len(data), sha256=hashlib.sha256(data).hexdigest(),
                failed=None, raw=data.decode())


def record(c, role, owner, pid, group, argv, raw=b"", *, exit_code=0, terminal=False, omit_raw=False):
    status = -exit_code if exit_code < 0 else exit_code << 8
    value = dict(role=role, owner=owner, pid=pid, group=group, status=status, exitCode=exit_code,
                 authorized=False, terminal=terminal, unknown=False, closed=[True, True, True],
                 waits=[dict(pid=0, status=0, count=1), dict(pid=pid, status=status)], argv=argv,
                 stdout=capture(raw), stderr=capture(b""))
    if omit_raw:
        value["stdout"].pop("raw")
        value["stderr"].pop("raw")
    return value

def refresh_captures(args):
    """Reserialize independent bytes so semantic negatives cannot rely on stale hashes.

    The fictional inside payload remains independent. Inner relay forwarding
    metadata is recomputed from the explicitly retained frame sequence outward.
    """
    frames = args[4]
    route = C.ROUTES[args[0]["row"]]
    if route.entry == "ruby_relay.rb" or "multi-hop" in route.name:
        raw = b"".join(C.dumps(v["frame"]) for v in frames
                       if v["stream"] == "route.stdout" and v["frame"]["actor"] in {"sender", "ruby-fork"})
        for value in frames:
            frame = value["frame"]
            if frame["actor"] == "ruby-relay" and frame["kind"] in {"final", "loss-ready", "loss-exit"}:
                frame["data"]["streams"][0].update(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())
    if route.entry == "python_relay.py":
        raw = b"".join(C.dumps(v["frame"]) for v in frames
                       if v["stream"] == "route.stdout" and v["frame"]["actor"] != "python-relay")
        for value in frames:
            frame = value["frame"]
            if frame["actor"] == "python-relay" and frame["kind"] == "final":
                frame["data"]["child"]["stdout"].update(observedBytes=len(raw), retainedBytes=len(raw),
                                                         sha256=hashlib.sha256(raw).hexdigest())
                frame["data"]["forwarded"]["stdout"] = len(raw)
    for index, label in ((0, "outside.stdout"), (1, "route.stdout")):
        raw = b"".join(C.dumps(v["frame"]) for v in frames if v["stream"] == label)
        args[2][index]["stdout"] = capture(raw)
    if route.loss:
        finals = [v["frame"] for v in frames if v["frame"]["actor"] == "sender" and v["frame"]["kind"] == "final"]
        commits = [v["frame"] for v in frames if v["frame"]["actor"] == "sender" and v["frame"]["kind"] == "terminal-commit"]
        if len(finals) == len(commits) == 1:
            final_raw = C.dumps(finals[0])
            commits[0]["data"] = dict(finalSha256=hashlib.sha256(final_raw).hexdigest(),
                                      finalSequence=finals[0]["sequence"], finalBytes=len(final_raw))
        args[5] = capture(b"".join(C.dumps(v["frame"]) for v in frames if v["stream"] == "independent"))
    return args

class SelectorFiles:
    """Fictional NAMED leaves only; all authority parsing/consumers stay real.

    The copied controls supply original filenames/shape, never access to their
    source/index payloads. Hash pins are independently recomputed for fictional
    content. Tests also check the unmodified actual control bytes separately.
    No selected pathname is ever opened by this fixture; unlisted reads fail.
    """
    def __init__(self, world):
        self.world, self.stack = world, ExitStack()
        self.snapshot = C.loads((BASE / "selector-control-snapshots/SNAPSHOT.json").read_bytes(), file_record=True)
        self.baseline = C.loads((BASE / "selector-control-snapshots/BASELINE.json").read_bytes(), file_record=True)
        self.files, self.modes, self.reads, self.on_read = {}, {}, [], None
        self.evidence, self.evidence_constructions = MemoryEvidence(), 0
        self.control_paths = tuple(row[0] for row in D.SELECTOR_CONTROLS)
        for prefix, rows in ((D.SOURCE, self.snapshot["files"]), (D.ROOT, self.baseline["rootUserFiles"])):
            for name, row in rows.items():
                path = str(prefix / name)
                payload = ("fictional selected bytes only: " + path + "\n").encode()
                self.files[path] = payload
                self.modes[path] = D.stat.S_IFREG | (0o700 if row["executable"] else 0o600)
                row["sha256"] = hashlib.sha256(payload).hexdigest()
        self.agents_sha = hashlib.sha256(self.files[str(D.ROOT / "AGENTS.md")]).hexdigest()
        self.snapshot["userAgentsSha256"] = self.baseline["rootAgentalSha256"] = self.agents_sha
        indexes = []
        for name, path, _sha in D.INDEX_ROWS:
            payload = ("fictional fixed index: " + name + "\n").encode()
            self.files[path] = payload
            sha = hashlib.sha256(payload).hexdigest()
            self.baseline["indexBindings"][name] = dict(path=path, sha256=sha)
            indexes.append((name, path, sha))
        self.index_rows = tuple(indexes)
        self.selected_paths = frozenset(self.files)
        self.script_payloads = {name: ("fictional code: " + name).encode() for name in C.SCRIPTS}
        self.files.update({str(BASE / name): raw for name, raw in self.script_payloads.items()})
        self.extra_reference = str(D.QA / "zz-fictional-approved-reference.txt")
        self.files[self.extra_reference] = b"fictional final reference barrier\n"
        self.seal()
        self.allowed_paths = frozenset(self.files)

    def seal(self, *, snapshot_raw=None, baseline_raw=None, bind_snapshot=True):
        """Freeze fictional pins BEFORE entry; this is not a runtime rebaseline."""
        first = C.dumps(self.snapshot) if snapshot_raw is None else snapshot_raw
        if bind_snapshot and type(self.baseline.get("bindings")) is dict:
            self.baseline["bindings"]["implementation-r6/SNAPSHOT.json"] = hashlib.sha256(first).hexdigest()
        second = C.dumps(self.baseline) if baseline_raw is None else baseline_raw
        self.controls = tuple((path, len(raw), hashlib.sha256(raw).hexdigest())
                              for path, raw in zip(self.control_paths, (first, second)))
        self.files.update(dict(zip(self.control_paths, (first, second))))
        self.references = {path: dict(bytes=size, sha256=sha) for path, size, sha in self.controls}
        extra = self.files[self.extra_reference]
        self.references[self.extra_reference] = dict(bytes=len(extra), sha256=hashlib.sha256(extra).hexdigest())
        self.bind_references()

    def bind_references(self, *, raw=None):
        references = C.dumps(self.references) if raw is None else raw
        self.files[str(BASE / "REFERENCE-BINDINGS.json")] = references
        self.approval = dict(decision="APPROVED_SINGLE_INHERITANCE_FEASIBILITY_EXECUTION", planSha256=C.PLAN_SHA,
                             scripts={name: hashlib.sha256(raw).hexdigest() for name, raw in self.script_payloads.items()},
                             matrixSha256=C.matrix_hash(), referenceBindingsSha256=hashlib.sha256(references).hexdigest())
        self.files[str(D.APPROVAL)] = C.dumps(self.approval)

    def read(self, path, cap=2 * 1024 * 1024):
        preparation_bound()
        name = str(path)
        self.reads.append(name)
        if name not in self.files:
            if name in self.allowed_paths:
                raise FileNotFoundError(errno.ENOENT, "fictional-missing-named-input", name)
            raise AssertionError("unlisted fictional input read: " + name)
        value = self.files[name]
        C.need(type(value) is bytes and len(value) <= cap, "fictional-regular-byte-cap")
        if self.on_read:
            self.on_read(name)
        return value

    def lstat(self, path):
        preparation_bound()
        name = str(path)
        if name not in self.modes:
            raise AssertionError("unlisted fictional leaf metadata: " + name)
        return types.SimpleNamespace(st_mode=self.modes[name])

    def evidence_factory(self):
        self.evidence_constructions += 1
        self.evidence_read_count = len(self.reads)
        return self.evidence

    def __enter__(self):
        for key, value in (("SELECTOR_CONTROLS", self.controls), ("INDEX_ROWS", self.index_rows),
                           ("AGENTS_SHA", self.agents_sha), ("TOOL_HASHES", {})):
            self.stack.enter_context(mock.patch.object(D, key, value))
        self.stack.enter_context(mock.patch.object(D, "regular_bytes", side_effect=self.read))
        self.stack.enter_context(mock.patch.object(Path, "lstat", lambda path: self.lstat(path)))
        self.stack.enter_context(mock.patch.object(D, "Cancellation", return_value=self.world.cancel))
        self.stack.enter_context(mock.patch.object(D, "Evidence", side_effect=self.evidence_factory))
        self.stack.enter_context(mock.patch.object(os.path, "lexists", return_value=False))
        self.stack.enter_context(mock.patch.object(sys, "platform", "darwin"))
        self.stack.enter_context(mock.patch.object(os, "uname", return_value=types.SimpleNamespace(
            sysname="fictional", release="1", version="1", machine="fixture")))
        return self

    def __exit__(self, *_args):
        self.stack.close()

    def selected_reads(self):
        return [name for name in self.reads if name in self.selected_paths]

class FictionalOutput:
    def __init__(self, on_frame=None):
        self.pending, self.frames, self.on_frame = {}, [], on_frame

    def write(self, fd, raw):
        preparation_bound()
        data = bytes(raw)
        value = self.pending.setdefault(fd, bytearray())
        value.extend(data)
        while b"\n" in value:
            line, _, tail = value.partition(b"\n")
            value[:] = tail
            frame = C.loads(bytes(line))
            self.frames.append((fd, frame))
            if self.on_frame:
                self.on_frame(fd, frame)
        return len(data)

class LossPublicationIO(FictionalWorld):
    """Real producer/encoder/pipe-write control flow, fictional syscall leaves.

    The root and sender have SEPARATE cancellation flags. Captured bytes are fed
    unchanged to real Reader/pump/cleanup/preproof/QueryChild/reconcile consumers.
    No fake claims a full atomic commit was written on an error/short return.
    Unsupported short/odd atomic returns are defensive-contract cases only.
    """
    def __init__(self, row_name, fault="none"):
        super().__init__(immediate_exit=True)
        self.args, self.fault = fixture(row_name), fault
        self.sender_cancel = FictionalCancellation()
        self.sender_checks, self.checks_after_commit = 0, 0
        self.result_raw, self.write_attempts, self.commit_attempts, self.exits = bytearray(), [], [], []
        self.pipe_queries, self.completed_final, self.committed = [], False, False
        self.armed, self.nonblocking, self.provisional_attempts = False, False, 0
        self.setter_calls, self.terminal_selects = 0, 0
        self.select_eagain_seen = False
        self._loss_stack = ExitStack()

    def sender_checkpoint(self):
        preparation_bound()
        self.sender_checks += 1
        if self.committed:
            self.checks_after_commit += 1
        self.sender_cancel.check()

    def stat(self, fd):
        if fd in (34, 81):
            if self.armed and fd == 34 and self.fault == "custody-error":
                raise OSError(errno.EIO, "fictional-result-fstat")
            inode = 99 if self.armed and (self.fault == "wrong-custody"
                       or self.fault == "commit-custody" and self.completed_final) else 7
            return types.SimpleNamespace(st_dev=5, st_ino=inode, st_mode=stat_mode_fifo(), st_rdev=0,
                                         st_uid=501, st_nlink=1, st_size=0, st_mtime_ns=1, st_ctime_ns=1)
        return super().stat(fd)

    def flags(self, fd, operation, *_args):
        if fd == 34:
            preparation_bound()
            C.need(operation == fcntl.F_GETFL, "fictional-result-fcntl")
            if self.armed and self.fault == "flags-error":
                raise OSError(errno.EIO, "fictional-result-fcntl")
            direction = os.O_RDONLY if self.armed and self.fault == "wrong-direction" else os.O_WRONLY
            blocked = self.armed and (self.fault == "missing-nonblocking"
                                     or self.fault == "commit-blocking" and self.completed_final)
            return direction | (os.O_NONBLOCK if self.nonblocking and not blocked else 0)
        return super().flags(fd, operation)

    def set_blocking(self, fd, value):
        if fd == 34:
            preparation_bound()
            self.setter_calls += 1
            C.need(value is False, "fictional-result-nonblocking-setting")
            if self.armed and self.fault == "set-blocking-error":
                raise OSError(errno.EIO, "fictional-set-blocking")
            self.nonblocking = True
            return
        return super().set_blocking(fd, value)

    def pipe_buf(self, fd, name):
        preparation_bound()
        self.pipe_queries.append((fd, name))
        C.need((fd, name) == (34, "PC_PIPE_BUF"), "fictional-pipebuf-owned-fd-only")
        if self.fault == "pipebuf-error":
            raise OSError(errno.EINVAL, "fictional-pipebuf-unsupported")
        if self.fault == "admission-pipebuf-eagain" and self.completed_final and len(self.pipe_queries) == 3:
            raise BlockingIOError(errno.EAGAIN, "fictional-metadata-is-not-commit-write")
        values = {"pipebuf-small": 511, "pipebuf-bool": True, "pipebuf-string": "512", "pipebuf-negative": -1,
                  "per-fd-pipebuf": 511}
        return values.get(self.fault, 512)

    def select(self, readers, writers, errors, timeout):
        if self.armed and writers == [34]:
            self.terminal_selects += 1
            if self.fault in {"cancel-select", "cancel-backpressure"} and self.provisional_attempts == 0:
                self.sender_cancel.cancelled = True
                return readers, [] if self.fault == "cancel-backpressure" else writers, errors
            if self.fault == "cancel-commit-select" and self.completed_final:
                self.sender_cancel.cancelled = True
                return readers, writers, errors
            if self.fault == "commit-select-eagain" and self.completed_final and not self.select_eagain_seen:
                self.select_eagain_seen = True
                raise BlockingIOError(errno.EAGAIN, "fictional-select-is-not-commit-write")
        return super().select(readers, writers, errors, timeout)

    def write_result(self, fd, raw):
        preparation_bound()
        C.need(fd == 34, "fictional-result-write-owned-fd-only")
        data = bytes(raw)
        is_commit = data.endswith(b"\n") and data.startswith(b'{"actor":"sender","data":{"finalBytes":')
        self.write_attempts.append((is_commit, data))
        if not self.armed:
            self.result_raw.extend(data)
            return len(data)
        if is_commit:
            self.commit_attempts.append(data)
            if self.fault in {"commit-eagain", "cancel-commit-eagain", "deadline-commit-eagain"} and len(self.commit_attempts) == 1:
                if self.fault == "cancel-commit-eagain":
                    self.sender_cancel.cancelled = True
                elif self.fault == "deadline-commit-eagain":
                    self.at = self.args[0]["disposition"]
                raise BlockingIOError(errno.EAGAIN, "fictional-zero-byte-backpressure")
            if self.fault == "commit-eagain-partial":
                self.result_raw.extend(data[:1])
                raise BlockingIOError(errno.EAGAIN, "fictional-unsupported-reported-partial", 1)
            if self.fault == "commit-error":
                raise OSError(errno.EIO, "fictional-no-commit-write")
            if self.fault == "commit-eintr":
                raise InterruptedError(errno.EINTR, "fictional-returned-EINTR-no-user-retry")
            if self.fault == "commit-short":
                self.result_raw.extend(data[:7])
                return 7
            if self.fault in {"commit-zero", "commit-bool", "commit-string", "commit-overcount"}:
                return {"commit-zero": 0, "commit-bool": True, "commit-string": "whole",
                        "commit-overcount": len(data) + 1}[self.fault]
            # Cancellation during a successfully ADMITTED whole API invocation
            # is intentionally irreversible. No internal kernel retry is tested.
            if self.fault == "cancel-admitted-commit":
                self.sender_cancel.cancelled = True
            if self.fault == "root-cancel-commit":
                self.cancel.cancelled = True
            if self.fault == "root-cutoff-commit":
                self.at = self.args[0]["outer"]
            self.result_raw.extend(data)
            self.committed = True
            return len(data)
        self.provisional_attempts += 1
        if self.fault == "provisional-error" or self.fault == "provisional-partial-error" and self.provisional_attempts > 1:
            raise OSError(errno.EIO, "fictional-partial-final-write")
        if self.fault in {"provisional-zero", "provisional-bool", "provisional-string", "provisional-overcount"}:
            return {"provisional-zero": 0, "provisional-bool": True, "provisional-string": "whole",
                    "provisional-overcount": len(data) + 1}[self.fault]
        if self.fault == "provisional-partial-error":
            self.result_raw.extend(data[:7])
            return 7
        if self.fault == "provisional-short-success" and self.provisional_attempts == 1:
            self.result_raw.extend(data[:7])
            return 7
        self.result_raw.extend(data)
        if self.fault == "cancel-between-chunks" and self.provisional_attempts == 1:
            self.sender_cancel.cancelled = True
        if data.endswith(b"\n"):
            self.completed_final = True
            if self.fault == "cancel-after-final":
                self.sender_cancel.cancelled = True
            if self.fault == "deadline-after-final":
                self.at = self.args[0]["disposition"]
        return len(data)

    def own_exit(self, code):
        self.exits.append(code)
        raise FictionalOwnExit(code)

    def __enter__(self):
        super().__enter__()
        self._loss_stack.enter_context(mock.patch.object(os, "write", side_effect=self.write_result))
        self._loss_stack.enter_context(mock.patch.object(os, "fpathconf", side_effect=self.pipe_buf))
        self._loss_stack.enter_context(mock.patch.object(os, "_exit", side_effect=self.own_exit))
        return self

    def __exit__(self, *args):
        self._loss_stack.close()
        super().__exit__(*args)

    def publish(self, *, code=0, legacy=None, value_change=None):
        self.owner_pid = 401
        c = self.args[0]
        guard = R.WorkGuard(c, types.SimpleNamespace(check=self.sender_checkpoint))
        emitter = R.Emitter(c, "sender", 34, guard=guard)
        # Use the real Emitter even for the pre-terminal prefix. The subsequent
        # expected frames are obtained by Reader decoding actual retained bytes.
        for item in self.args[4]:
            frame = item["frame"]
            if item["stream"] == "independent" and frame["kind"] not in {"final", "terminal-commit"}:
                emitter.emit(frame["kind"], frame["data"])
        value = copy.deepcopy(sender_final(self.args))
        if value_change:
            value_change(value)
        if legacy:
            value["terminal"]["mode"] = "kernel-close-on-immediate-exit"
        self.armed = True
        if self.fault == "cancel-before-final":
            self.sender_cancel.cancelled = True
        try:
            if legacy:
                legacy["publish_loss"](emitter, value, code, guard.completion)
            else:
                S.publish_loss(emitter, value, code)
        except FictionalOwnExit:
            pass
        else:
            raise AssertionError("loss publisher returned instead of immediate own exit")
        self.owner_pid = 100
        C.need(len(self.exits) == 1, "fictional-one-own-exit")
        return bytes(self.result_raw)


def root_loss_row(world, raw, *, eof=True, read_error=False, args=None):
    """No mocked root consumer. Real typed children, captures, cleanup and pump."""
    args = copy.deepcopy(world.args if args is None else args)
    row = world.row(args[0]["row"])
    row.c, row.registry.c, row.owner.identity = args[0], args[0], args[1]
    row.owner.acquisition_reserved, row.owner.exec_reserved = row.route.acquisitions, row.route.execs
    row.root_signals, row.sender_pid, row.subtree = args[7], 401, "unresolved"
    row.registry.attempts = args[3][:2]
    for index, (cls, value) in enumerate(zip((R.OutsideChild, R.RelayChild), args[2][:2])):
        streams = [FictionalStream(world, 900 + index * 10 + offset) for offset in range(3)]
        world.streams.extend(streams)
        world.directions[streams[0].fd] = os.O_WRONLY
        handle = types.SimpleNamespace(pid=value["pid"], stdin=streams[0], stdout=streams[1], stderr=streams[2],
                                       returncode=value["exitCode"])
        child = cls(row.registry)
        row.registry.children.append(child)
        child.publish(handle)
        child.adopt(handle, row.owner.capture)
        child.argv, child.group = value["argv"], value["group"]
        child.status, child.terminal = value["status"], value["terminal"]
        child.custody_known, child.authorized, child.waits = False, False, copy.deepcopy(value["waits"])
        for reader, label in ((child.out, "stdout"), (child.err, "stderr")):
            data = value[label]["raw"].encode()
            world.read_sequences[reader.fd] = ([data] if data else []) + [b""]
        if index == 0:
            row.outside = child
        else:
            row.relay = child
    reader = R.Reader(81, row.owner.capture)
    world.read_sequences[81] = [raw] + ([OSError(errno.EIO, "fictional-R-read-failure")] if read_error else [b""] if eof else [])
    row.channels.reader = reader
    row.channels.entries.append(dict(fd=81, role="r-read-root", direction=os.O_RDONLY, identity=reader.identity,
                                      closed=False, closeAttempted=False, closeError=None))
    row.cleanup()  # Real Reader zero-byte EOF, never a parent-side close inference.
    return row


def row_reconcile_args(row):
    return [row.c, row.owner.identity, [child.record() for child in row.registry.children],
            row.registry.attempts, row.frames, row.channels.reader.record(), row.absent,
            row.root_signals, row.observations]


def prove_root_loss(test, world, raw, *, accepted, eof=True, read_error=False, old=None, args=None):
    row = root_loss_row(world, raw, eof=eof, read_error=read_error, args=args)
    entry_counts = row.owner.acquisition_reserved, row.owner.exec_reserved
    if accepted:
        # The only old-consumer replacement is the explicit, separately labelled
        # historical mutant control. All corrected paths use the actual functions.
        with ExitStack() as stack:
            if old:
                stack.enter_context(mock.patch.object(D, "route_terminal", old["route_terminal"]))
            preproof = row.terminal_preproof()
            test.assertEqual(preproof["subtreeDisposition"], "awaiting-absence")
            test.assertTrue(row.unknown)
            test.assertEqual(world.launches, [])
            row.query_orphan()
        proof = (old["reconcile"] if old else V.reconcile)(*row_reconcile_args(row))
        test.assertEqual(proof["subtreeDisposition"], "resolved")
        test.assertEqual(proof["senderExitStatus"], "not-waitable-by-root")
        test.assertEqual((row.owner.acquisition_reserved, row.owner.exec_reserved),
                         (entry_counts[0] + 1, entry_counts[1] + 1))
        test.assertEqual(len(world.launches), 1)
        test.assertEqual(world.wait_statuses, [(601, 256)])
        test.assertIs(row.registry.children[-1].handle, world.handle)
        test.assertTrue(all(row.registry.children[-1].closed))
        test.assertFalse(row.unknown)
    else:
        with test.assertRaises(C.Failure):
            row.terminal_preproof()
        with test.assertRaises(C.Failure):
            row.query_orphan()
        with test.assertRaises((C.Failure, ValueError)):
            V.reconcile(*row_reconcile_args(row))
        test.assertEqual((row.owner.acquisition_reserved, row.owner.exec_reserved), entry_counts)
        test.assertEqual(world.launches, [])
        test.assertEqual(world.wait_statuses, [])
        test.assertEqual(row.observations, [])
        test.assertTrue(row.unknown)
        # Exercise the actual deletion admission guard: no successful scratch
        # disposal can follow a failed/missing final/commit/EOF/absence proof.
        with test.assertRaises(C.Failure):
            row.archive_and_remove_scratch()
    test.assertEqual(world.signals, [])
    return row

        cases = ("missing", "duplicate", "reversed", "old-mode", "digest", "length", "final-sequence", "sequence",
                 "length-bool", "sequence-bool", "digest-type", "extra-data", "missing-data", "wrong-row", "nonce-replay",
                 "wrong-pid", "wrong-work", "wrong-disposition", "wrong-outer", "wrong-channel", "trailing",
                 "noncanonical-final", "noncanonical-commit", "crlf-commit", "truncated", "raw-over512")
        for name in ("parent-exit", "parent-kill"):
            for case in cases:
                with self.subTest(route=name, case=case), LossPublicationIO(name) as world:
                    original = world.publish()
                    lines = original.splitlines(keepends=True)
                    events = [C.loads(line) for line in lines]
                    final, commit = events[-2:]
                    args = None
                    if case == "missing":
                        events.pop()
                    elif case == "duplicate":
                        events.append(dict(copy.deepcopy(commit), sequence=commit["sequence"] + 1))
                    elif case == "reversed":
                        events[-2:] = [commit, final]
                        commit["sequence"], final["sequence"] = final["sequence"], commit["sequence"]
                        commit["data"]["finalSequence"] = final["sequence"]
                    elif case == "old-mode":
                        final["data"]["terminal"]["mode"] = "kernel-close-on-immediate-exit"
                        final_bytes = C.dumps(final)
                        commit["data"].update(finalSha256=hashlib.sha256(final_bytes).hexdigest(), finalBytes=len(final_bytes))
                    elif case in {"digest", "length", "final-sequence", "length-bool", "sequence-bool", "digest-type", "raw-over512"}:
                        key, value = {"digest": ("finalSha256", "0" * 64), "length": ("finalBytes", len(lines[-2]) + 1),
                                      "final-sequence": ("finalSequence", final["sequence"] - 1),
                                      "length-bool": ("finalBytes", True), "sequence-bool": ("finalSequence", True),
                                      "digest-type": ("finalSha256", 1), "raw-over512": ("finalBytes", 10 ** 240)}[case]
                        commit["data"][key] = value
                    elif case == "sequence":
                        commit["sequence"] += 1
                    elif case == "extra-data":
                        commit["data"]["extra"] = True
                    elif case == "missing-data":
                        del commit["data"]["finalBytes"]
                    elif case in {"wrong-row", "nonce-replay", "wrong-pid", "wrong-work", "wrong-disposition", "wrong-outer"}:
                        key, value = {"wrong-row": ("row", "parent-kill" if name == "parent-exit" else "parent-exit"),
                                      "nonce-replay": ("nonce", "b" * 32), "wrong-pid": ("pid", 402),
                                      "wrong-work": ("work", 131.0), "wrong-disposition": ("disposition", 141.0),
                                      "wrong-outer": ("outer", 281.0)}[case]
                        commit[key] = value
                    elif case == "trailing":
                        events.append(dict(copy.deepcopy(commit), sequence=commit["sequence"] + 1,
                                           kind="query", data=copy.deepcopy(final["data"]["queries"][0])))
                    raw = b"".join(C.dumps(frame) for frame in events)
                    if case == "wrong-channel":
                        args = copy.deepcopy(world.args)
                        args[2][1]["stdout"] = capture(args[2][1]["stdout"]["raw"].encode() + lines[-1])
                        raw = b"".join(lines[:-1])
                    elif case == "noncanonical-final":
                        altered = b" " + lines[-2]
                        commit["data"].update(finalSha256=hashlib.sha256(altered).hexdigest(), finalBytes=len(altered))
                        raw = b"".join(lines[:-2]) + altered + C.dumps(commit)
                    elif case == "noncanonical-commit":
                        raw = b"".join(lines[:-1]) + b" " + lines[-1]
                    elif case == "crlf-commit":
                        raw = original[:-1] + b"\r\n"
                    elif case == "truncated":
                        raw = original[:-1]
                    row = prove_root_loss(self, world, raw, accepted=False, args=args)
                    if case == "raw-over512":
                        self.assertGreater(len(C.dumps(commit)), 512)
                        self.assertTrue(row.clean)
                        with self.assertRaises(C.Failure) as caught:
                            row.terminal_preproof()
                        self.assertEqual(caught.exception.code, "terminal-commit-raw-byte-bound")

    def test_immutable_authority_reuses_original_rows_and_final_selector_drift_rejects_before_reads(self):
        for phase in ("mutable-dicts-only", "before-final", "after-final-verification"):
            with self.subTest(phase=phase), FictionalWorld() as world, SelectorFiles(world) as files:
                controller = D.Controller(100.0)
                controller.prepare()
                authority = controller.selectors
                with self.assertRaises(AttributeError):
                    authority.source_rows = ()
                with self.assertRaises(TypeError):
                    authority.source_rows[0][0] = "replaced"
                files.snapshot["files"].clear()
                files.baseline["indexBindings"]["root"]["path"] = "/fictional/unapproved-index"
                files.reads.clear()
                if phase == "before-final":
                    files.files[files.control_paths[0]] = b"{}\n"
                if phase == "after-final-verification":
                    def after_read(name):
                        if name == files.control_paths[1]:
                            files.files[files.control_paths[0]] = b"{}\n"
                            files.files[files.control_paths[1]] = b"{}\n"
                    files.on_read = after_read
                if phase == "before-final":
                    with self.assertRaises(C.Failure):
                        controller.final_preservation()
                    self.assertEqual(files.selected_reads(), [])
                else:
                    self.assertEqual(controller.final_preservation(), controller.before)
                    self.assertIs(controller.selectors, authority)
                    self.assertEqual(set(files.selected_reads()), files.selected_paths)
                    self.assertNotIn("/fictional/unapproved-index", files.reads)
                self.assertEqual((world.launches, world.signals, world.wait_calls), ([], [], 0))
