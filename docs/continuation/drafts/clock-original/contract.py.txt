class Failure(Exception):
    def __init__(self, code, **facts):
        super().__init__(code)
        self.code, self.facts = code, facts


def need(value, code, **facts):
    if not value:
        raise Failure(code, **facts)


def number(value):
    return type(value) in (int, float) and math.isfinite(value)


def positive_pid(value):
    return type(value) is int and value > 1


def dumps(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode() + b"\n"


def loads(raw, *, file_record=False):
    need(type(file_record) is bool, "json-role")
    need(type(raw) is bytes and len(raw) <= (2 * 1024 * 1024 if file_record else STREAM_CAP), "json-size")

    def pairs(items):
        result = {}
        for key, value in items:
            need(key not in result, "json-duplicate-key", key=key)
            result[key] = value
        return result

    def finite_float(value):
        result = float(value)
        need(math.isfinite(result), "json-nonfinite")
        return result

    def bad_constant(value):
        raise Failure("json-nonfinite", value=value)

    result = json.loads(raw, object_pairs_hook=pairs, parse_float=finite_float,
                        parse_constant=bad_constant)
    stack, nodes = [(result, 0)], 0
    while stack:
        value, depth = stack.pop()
        nodes += 1
        need(nodes <= (32768 if file_record else 8192) and depth <= 20, "json-structure-bound")
        if type(value) is dict:
            stack.extend((v, depth + 1) for v in value.values())
        elif type(value) is list:
            stack.extend((v, depth + 1) for v in value)
    return result

def matrix():
    return [dict(name=r.name, entry=r.entry, acquisitions=r.acquisitions, execs=r.execs,
                 tokens=list(r.tokens), fenced=r.fenced, loss=r.loss) for r in ROWS]


def matrix_hash():
    return hashlib.sha256(dumps(matrix())).hexdigest()

        return [PYTHON, "-I", "-B", str(BASE / "canary.py"), context, token, str(owner)]
    if actor == "root" and token == "route":

                   [PYTHON, "-I", "-B", str(BASE / entry), context])

def argv_matches(actual, expected, c):
    need(type(actual) is list and len(actual) == len(expected)
         and all(type(s) is str for s in actual), "argv-shape")
    for found, wanted in zip(actual, expected):
        if wanted == dumps(c).decode().strip():
            need(loads(found.encode()) == c, "argv-context")
        else:
            need(found == wanted, "argv-literal")


def policy(marker_path, fenced):
    need(type(fenced) is bool and re.fullmatch(r"/[A-Za-z0-9_./-]+", str(marker_path)), "policy-literal")
    return (POLICY_PREFIX + (SIGNAL_RULE if fenced else "")
            + f'(deny file-read-data (literal "{marker_path}"))\n').encode()


def validate_context(c):
    keys = {"row", "nonce", "rootPid", "uid", "start", "work", "disposition", "outer",
            "scratch", "markerSha256", "policySha256", "scripts", "external", "channels"}
    need(type(c) is dict and set(c) == keys, "context-shape")
    need(type(c["row"]) is str and c["row"] in ROUTES and type(c["nonce"]) is str
         and re.fullmatch(r"[0-9a-f]{32}", c["nonce"]), "context-row-nonce")
    need(positive_pid(c["rootPid"]) and type(c["uid"]) is int and c["uid"] >= 0, "context-identity")
    need(all(number(c[k]) for k in ("start", "work", "disposition", "outer")), "context-time-type")
    need(c["work"] == c["start"] + WORK and c["disposition"] == c["work"] + DISPOSITION
         and c["disposition"] <= c["outer"], "context-deadline-reset")
    need(c["scratch"] == str(BASE / "execution-r1" / (c["row"] + "-inputs")), "context-scratch")
    for key in ("markerSha256", "policySha256"):
        need(type(c[key]) is str and re.fullmatch(r"[a-f0-9]{64}", c[key]), "context-hash")
    need(type(c["scripts"]) is dict and set(c["scripts"]) == set(SCRIPTS)
         and all(type(v) is str and re.fullmatch(r"[a-f0-9]{64}", v) for v in c["scripts"].values()), "context-scripts")
    need(hashlib.sha256(policy(Path(c["scratch"]) / "marker", ROUTES[c["row"]].fenced)).hexdigest()
         == c["policySha256"], "context-policy")
    ext = c["external"]
    need(ext is None or (type(ext) is dict and set(ext) == {"pid", "group", "parent", "uid"}
                         and all(type(ext[k]) is int for k in ext)
                         and positive_pid(ext["pid"]) and ext["pid"] == ext["group"]
                         and ext["parent"] == c["rootPid"] and ext["uid"] == c["uid"]), "context-external")
    ch = c["channels"]
    need(ch is None or (ROUTES[c["row"]].loss and type(ch) is dict
                       and set(ch) == {"c", "r", "cIdentity", "rIdentity"}), "context-channels")
    if ch is not None:
        need(type(ch["c"]) is int and type(ch["r"]) is int and ch["c"] > 2
             and ch["r"] > 2 and ch["c"] != ch["r"], "context-channel-fds")
        for k in ("cIdentity", "rIdentity"):
            need(type(ch[k]) is list and len(ch[k]) == 4
                 and all(type(v) is int and v >= 0 for v in ch[k]), "context-channel-identity")
    return c


def admit_row(now, outer, acquisitions, execs, route):
    need(number(now) and number(outer) and outer - now >= WORK + DISPOSITION, "row-time-admission")
    need(type(acquisitions) is int and acquisitions >= 0 and acquisitions + route.acquisitions <= ACQUISITION_CAP,
         "row-acquisition-admission")
    need(type(execs) is int and execs >= 0 and execs + route.execs <= EXEC_CAP, "row-exec-admission")


def admit_query(now, c, count, acquisitions, execs, capture_available):
    need(ROUTES[c["row"]].loss and type(count) is int and 0 <= count < 3, "query-row-count")
    need(number(now) and min(c["disposition"], c["outer"]) - now >= QUERY_WORK + QUERY_CLEANUP,
         "query-original-time-admission")
    need(type(acquisitions) is int and type(execs) is int and 0 <= acquisitions < ACQUISITION_CAP
         and 0 <= execs < EXEC_CAP, "query-count-admission")
    need(type(capture_available) is int and capture_available >= 2 * STREAM_CAP, "query-byte-admission")


@dataclass(frozen=True)
class QueryWindow:
    """The original observer reserve, never a caller-replaceable signal timeout."""
    admitted: float
    work: float
    disposition: float

    @classmethod
    def original(cls, admitted, c):
        value = cls(admitted, admitted + QUERY_WORK, admitted + QUERY_WORK + QUERY_CLEANUP)
        value.validate(c)
        return value

    def validate(self, c):
        need(all(number(v) for v in (self.admitted, self.work, self.disposition))
             and c["start"] <= self.admitted and self.work == self.admitted + QUERY_WORK
             and self.disposition == self.work + QUERY_CLEANUP
             and self.disposition <= min(c["disposition"], c["outer"]), "query-window-binding")

    def record(self):
        return dict(admitted=self.admitted, work=self.work, disposition=self.disposition)

    need(value["contextSha256"] == hashlib.sha256(dumps(c)).hexdigest()
         and all(number(value[k]) and value[k] == c[k] for k in ("work", "disposition", "outer"))
         and number(value["admittedAt"]) and c["start"] <= value["admittedAt"] < min(c["work"], c["outer"]),
         "entry-original-context-window")
    need(value["controlRole"] == ("C" if actor == "sender" and ROUTES[c["row"]].loss else "stdin")
         and value["controlEof"] is False and value["controlFailed"] is None and value["cancelled"] is False,

def validate_frame(e, c, sequences):

    need(e["kind"] != "terminal-commit" or (e["actor"] == "sender" and ROUTES[c["row"]].loss is not None),

def marker_ok(observation, readable):
    need(type(observation) is dict and set(observation) == {"readable", "matches", "errno"}, "marker-shape")
    need(type(observation["readable"]) is bool and type(observation["matches"]) is bool, "marker-type")
    if readable:

             and type(observation["errno"]) is int and observation["errno"] in (errno.EPERM, errno.EACCES),
             "marker-not-permission-denied")

def query_result(stdout, stderr, exit_code, expected):
    need(type(stdout) is bytes and type(stderr) is bytes and type(exit_code) is int
         and type(expected) is dict and set(expected) == {"pid", "parent", "group"}
         and positive_pid(expected["pid"]) and expected["group"] == expected["pid"]
         and type(expected["parent"]) is int and expected["parent"] > 0, "query-result-shape")
    need(stderr == b"", "query-stderr")
    if exit_code == 1 and stdout.strip() == b"":
        return {"absent": True, "waitForSender": None, "method": "ps-empty-selection-exit1"}
    need(exit_code == 0, "query-failed-status", status=exit_code)
    lines = stdout.splitlines()
    need(len(lines) == 1, "query-selection-ambiguous")
    fields = lines[0].split()
    need(len(fields) == 4 and all(re.fullmatch(rb"[0-9]{1,10}", x) for x in fields[:3])
         # Installed /usr/share/man/man1/ps.1:316-383. A present state is
         # never an absence proof, including a zombie or an exiting process.
         and re.fullmatch(rb"[IRSTUZ][+<>AELNSsVWX]{0,15}", fields[3]), "query-parser")
    actual = dict(zip(("pid", "parent", "group"), map(int, fields[:3])))
    need(actual == expected, "query-reused-or-ambiguous", actual=actual, expected=expected)
    return {"absent": False, "waitForSender": None, "identity": actual,
            "stat": fields[3].decode("ascii"), "method": "ps-present-not-absence"}

def failure(exc):
    if isinstance(exc, Failure):
        return dict(type=type(exc).__name__, code=exc.code, facts=exc.facts)
    return dict(type=type(exc).__name__, code="unexpected", message=str(exc)[:2048])
