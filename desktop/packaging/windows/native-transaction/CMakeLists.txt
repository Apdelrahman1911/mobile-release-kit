cmake_minimum_required(VERSION 3.24)
project(MrkInstallerTransaction LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC required")
endif()
add_library(MrkInstallerTransaction SHARED transaction.cpp)
target_compile_features(MrkInstallerTransaction PRIVATE cxx_std_17)
target_compile_definitions(MrkInstallerTransaction PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(MrkInstallerTransaction PRIVATE /W4 /WX /guard:cf /sdl)
target_link_options(MrkInstallerTransaction PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
set_property(TARGET MrkInstallerTransaction PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(MrkInstallerTransaction PRIVATE "${MRK_BINDING_DIRECTORY}")
add_executable(MrkTransactionNativeTests native_tests.cpp)
target_compile_features(MrkTransactionNativeTests PRIVATE cxx_std_17)
target_compile_definitions(MrkTransactionNativeTests PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_include_directories(MrkTransactionNativeTests PRIVATE "${MRK_BINDING_DIRECTORY}")
set_property(TARGET MrkTransactionNativeTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
